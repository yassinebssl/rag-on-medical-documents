\begin{frame}{Topic and goal}
  \begin{itemize}
    \item Answer medical questions \alert{only} from WHO guidelines and PubMed OA articles
    \item Every claim cites its passage: \texttt{[source, p.\,N]}
    \item No good evidence $\Rightarrow$ fixed refusal sentence
    \item Goal: measure it against an LLM alone and a BM25 RAG
  \end{itemize}
  \vfill
  \small Student project. Not medical advice. Free LLMs only.
  \note{Speaker: Djerdali Mohamed Achraf. Why: an LLM alone answers from memory, and a wrong dose reads as confidently as a right one.}
\end{frame}

\begin{frame}{Dataset: 50 documents}
  \begin{tabular}{@{}lll@{}}
    \toprule
    Source & Count & Size \\
    \midrule
    WHO guidelines (PDF, WHO IRIS) & 20 & 25.9\,MB \\
    PubMed Central OA (full text) & 30 & 1.3\,MB \\
    \bottomrule
  \end{tabular}
  \bigskip
  \begin{itemize}
    \item Hypertension, diabetes, TB, malaria, HIV, antenatal care, paracetamol
    \item 1530 records (1500 WHO pages) $\to$ 3410 chunks
    \item Manifest: URL and size of every file, re-downloadable
  \end{itemize}
  \note{Speaker: Boussoualim Yassine. Sizes are the byte totals of data/raw/manifest.json. PubMed articles come as text because Europe PMC refuses PDFs to scripts. Two WHO PDFs are encrypted; pypdf reads them with cryptography.}
\end{frame}

\begin{frame}{Architecture (as built)}
  \centering
  \adjustbox{max width=\linewidth, max height=0.74\textheight}{\archfigure}
  \note{Speaker: Aissaoui Anes Safouane. Offline: chunk, embed with BGE-M3, store in Chroma; BM25 is built in memory from the same chunks. Online: hybrid retrieval, cross-encoder gate (MiniLM, threshold 0.5), free LLM, citation check; two ways to refuse.}
\end{frame}

\begin{frame}{Evaluation plan}
  \textbf{40 questions:} 20 answerable (A) \quad 10 out-of-corpus (O) \quad 10 held-out (N)
  \medskip

  \begin{tabular}{@{}ll@{}}
    \toprule
    System & What it tests \\
    \midrule
    \texttt{rag\_dense} & dense retrieval, cosine gate \\
    \texttt{rag\_hybrid} & + BM25 fused by RRF \\
    \texttt{rag\_rerank} & + cross-encoder gate (main) \\
    \texttt{bm25\_rag} & baseline: BM25 only, no gate \\
    \texttt{no\_rag} & baseline: LLM alone \\
    \bottomrule
  \end{tabular}
  \medskip

  \small Answer and refusal rates, hit rate, context precision, faithfulness, human rating
  \note{Speaker: Benahcene Mohamed Wassim. A questions have a ground truth, page and verbatim quote (20/20 found in the corpus). N questions are on corpus topics but ask facts the corpus lacks; written before any result. 50 free requests a day, so the LLM run takes several days.}
\end{frame}
